% FIGURE-EMBEDDING SWITCH

% MiMB_chapter_withfigures.tex issues \newif\ifshowfigures and

  \newif\ifshowfigures
  \showfiguresfalse
}{}
\makeatother

%------------------------------------------------------------------------------
% FIGURE HANDLING
%
% \FIGBOX{filename-without-extension}{width as fraction of textwidth}
%
% If the image file exists in the project, it is included (via
% \includegraphics with no extension - graphicx resolves it once existence
% is confirmed). If not, a labelled grey placeholder box is drawn instead,
% so the document always compiles and you can see exactly which figures are
% still missing. Only called when \ifshowfigures is true (see body.tex).
%
% Upload your PNGs to the Overleaf project's figures/ folder to replace the
% placeholders. Expected paths:
%     figures/fig1_workflow.png
%     figures/fig1_umap_celltypes.png
%     figures/fig2_tf_featureplots.png
%     figures/fig3_marker_dotplot.png
%     figures/fig4_networks_mature-endodermis.png
%     figures/fig4_networks_dividing_cells.png
%     figures/fig4_networks_exodermis.png
%------------------------------------------------------------------------------
\usepackage{tikz}
\newcommand{\FIGBOX}[2]{%
  \IfFileExists{#1.png}{\includegraphics[width=#2\textwidth]{#1}}{%
  \IfFileExists{#1.pdf}{\includegraphics[width=#2\textwidth]{#1}}{%
    \fbox{\begin{minipage}[c][0.34\textwidth][c]{#2\textwidth}%
      \centering\color{gray}%
      {\ttfamily\small\detokenize{#1}}\\[6pt]
      {\rmfamily\small image file not found in project}%
    \end{minipage}}}}%
}
